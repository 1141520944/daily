
## 快慢指针
```go

func removeElement(nums []int, val int) int {  
    s, f := 0, 0  
    for f < len(nums) {  
       if nums[f] != val {  
          nums[s] = nums[f]  
          s++  
       }  
       f++  
    }  
    return s  
}

```
## 滑动窗口

``` go
func minSubArrayLen(target int, nums []int) int {  
    l, r := 0, 0  
    sum := 0  
    res := math.MaxInt32  
  
    for r < len(nums) {  
       sum += nums[r] // 扩大窗口  
       for sum >= target { // 缩小窗口  
          res = min(res, r-l+1)  
          sum -= nums[l]  
          l++  
       }  
       r++  
    }  
  
    if res == math.MaxInt32 {  
       return 0  
    }  
    return res  
}

```
